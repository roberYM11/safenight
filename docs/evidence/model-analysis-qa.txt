base = Path("docs/model-analysis")

    with path.open() as f:
        return list(csv.DictReader(f))
def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()
bundle = json.loads(Path("cloud/app/server/data/manifest.json").read_text())
m = json.loads((base/"metrics.json").read_text())
snapshot = Path(m["snapshot"])
assert sha(snapshot) == m["snapshot_sha256"] == bundle["training_sha256"]
assert sha(Path("backend/scripts/compare_models.py")) == m["script_sha256"]
rows = read_rows(snapshot)
train = np.array([r["sensor_id"] in set(bundle["train_groups"]) for r in rows])
test = np.array([r["sensor_id"] in set(bundle["test_groups"]) for r in rows])
assert not np.any(train & test) and np.all(train | test)
assert (len(rows),len(set(r["sensor_id"] for r in rows)),train.sum(),test.sum()) == (3230,652,2586,644)
print("PASS snapshot and comparison script SHA-256, 3230 source rows, 652 sensors, exact manifest split 2586/644 rows and disjoint groups")
x = np.array([(float(r["lux"]),float(r["noise_db"])) for r in rows])
y = np.array([float(r["safety_index"]) for r in rows])
pdp = read_rows(base/"partial_dependence.csv")
metrics = read_rows(base/"metrics.csv")
for i,(name,leaf) in enumerate([("train_median",None),("current_leaf3",3),("variant_leaf1",1)]):
    if leaf is None:
        pred = np.full(sum(test),np.median(y[train]))
    else:
        model = RandomForestRegressor(n_estimators=114,max_depth=18,min_samples_leaf=leaf,random_state=42,n_jobs=1).fit(x[train],y[train])
        pred = model.predict(x[test])
        for col,key in enumerate(("lux","noise_db")):
            panel = [r for r in pdp if r["feature"]==key]
            grid = np.linspace(*np.percentile(x[train,col],[5,95]),51)
            np.testing.assert_allclose(grid,[float(r["value"]) for r in panel],rtol=0,atol=1e-12)
            library = partial_dependence(model,x[test],[col],method="brute",custom_values={col:grid})
            np.testing.assert_allclose(library.average[0],[float(r[name]) for r in panel],rtol=0,atol=1e-12)
            print("PASS",name,key,"all 51 values match sklearn brute PDP; endpoints",round(library.average[0][0],3),round(library.average[0][-1],3))
    error = y[test]-pred
    actual = (abs(error).mean(),np.sqrt(np.mean(error**2)),1-np.sum(error**2)/np.sum((y[test]-y[test].mean())**2))
    scores = m["metrics"][i]
    assert metrics[i]["model"]==scores["model"]==name
    for number,k in zip(actual,("mae","rmse","r2")):
        np.testing.assert_allclose([scores[k],float(metrics[i][k])],number,rtol=0,atol=1e-12)
        delta = number - m["metrics"][1][k]
        np.testing.assert_allclose([scores["delta_"+k],float(metrics[i]["delta_"+k])],delta,rtol=0,atol=1e-12)
    if leaf==3:
        np.testing.assert_allclose([actual[0],actual[2]],[bundle["validation_mae"],bundle["validation_r2"]],rtol=0,atol=1e-12)
    print("PASS",name,"manual MAE/RMSE/R2 and delta in both outputs",*[round(s,6) for s in actual])
print("PASS production reference MAE/R2, two-input PDPs, all independent tolerances <=1e-12")
